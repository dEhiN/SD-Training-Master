Create DB:
    CREATE DATABASE batch_8;

Select DB to use:
    USE batch_8;

Create table:
    CREATE TABLE users_info_table (
	    id INT PRIMARY KEY AUTO_INCREMENT,
        user_name VARCHAR(200),
        age INT,
        hobby VARCHAR(300),
        city VARCHAR(200)
    );
